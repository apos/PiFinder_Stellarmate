\subsection{Project summary \& focus}
% ============================================================

\begin{frame}[t]{What PFSM is}
\begin{itemize}
\item \textbf{A setup script} — installs \& patches PiFinder onto StellarMate OS.
\item \textbf{\code{diffs/*.diff} patches} — applied to a stock checkout via \code{patch(1)}.
\item \textbf{Three INDI drivers} — LX200, Mount Bridge, Simulator.
\item \textbf{A web Control Center} — stdlib-only, port 8765.
\end{itemize}
\end{frame}

\begin{frame}[t]{What PFSM is not}
\begin{itemize}
\item \textbf{Not a fork of PiFinder} — patches re-applied every update, kept minimal.
\item \textbf{Not a fork of StellarMate} — stock INDI, Web Manager, KStars.
\item \textbf{Not a replacement for the INDI Control Panel} — it stays for advanced work.
\item \textbf{Not PiFinder-version-specific} — talks only to the stable \code{/api/*}.
\end{itemize}
\end{frame}

\begin{frame}[t]{Why INDI}
\textbf{Both sides already speak it.}
\begin{itemize}
\item PiFinder exposes its position as an INDI telescope.
\item The Mount Bridge couples that to any INDI mount — no per-mount code.
\end{itemize}
\end{frame}

\begin{frame}[t]{Versions \& platforms}
\textbf{PiFinder 2.6.3 \quad$\cdot$\quad StellarMate OS 2.3.0}
\begin{itemize}
\item Pi 4 — fully tested.
\item Pi 5 — fully tested.
\item UTM x86 — install + Control Center + Mount Bridge vs.\ the Simulator (no real solving).
\end{itemize}
\medskip{}
Pinned to a fixed release tag, not the upstream \emph{release} branch.
\end{frame}

\begin{frame}[t]{Why a Control Center}
\begin{columns}[T]
\column{0.34\textwidth}
\begin{itemize}
\item Watch an install from a phone at the scope.
\item Switch Fake Mode $\leftrightarrow$ the real service.
\item Check camera / IMU / GPS are really detected.
\item Reboot the Pi without SSH.
\end{itemize}

\begin{frame}[t]{Stdlib only — by necessity}
\begin{columns}[T]
\column{0.34\textwidth}
\begin{itemize}
\item It bootstraps the very venv that PiFinder needs, so it can't depend on one.
\item \code{http.server} on \code{:8765}, polled every 1--2\,s.
\item HTTP Basic auth against the system account (PAM).
\end{itemize}

\subsection{PF LX200 \& Mount Bridge}
% ============================================================

\begin{frame}[t]{Three building blocks}
\begin{enumerate}
\item \textbf{PiFinder LX200} — INDI telescope. Reports the solved position; a GoTo becomes a push-to. Required.
\item \textbf{PiFinder Mount Bridge} — optional. Snoops PiFinder + the mount and couples them. Generic INDI only.
\item \textbf{Your mount's driver} — not ours. Only for a motorised mount.
\end{enumerate}
\medskip{}
\begin{center}
\includegraphics[width=0.82\linewidth,keepaspectratio]{mb_diagram}
\end{center}
\end{frame}

\begin{frame}[t]{The Web Manager — a ``Remote'' profile}
\begin{columns}[T]
\column{0.42\textwidth}
\begin{itemize}
\item PFSM's INDI drivers live \textbf{only} in the StellarMate Web Manager (\code{:8624}) — the Equipment Profile is built there.
\item KStars / Ekos then connects in \textbf{Remote Host} mode, never ``Local''.
\item The one real difference from a stock StellarMate setup.
\end{itemize}
\column{0.56\textwidth}
\hfill\includegraphics[width=\linewidth,height=0.78\textheight,keepaspectratio]{webmanager_full}
\end{columns}
\end{frame}

\begin{frame}[t]{One weak PiFinder, one strong box}
Pick the \textbf{role} at setup:
\smallskip{}
\begin{center}
\includegraphics[width=\linewidth,keepaspectratio]{cc_roles}
\end{center}
\medskip{}
\begin{itemize}
\item \textbf{Control host} — a strong StellarMate PC runs INDI, Ekos, guiding and capture; the low-power PiFinder Pi over the LAN only supplies its solved position.
\item \textbf{INDI-only install} (\code{--mode=indi\_only}) — the Control-host case: just the INDI deps + the two PFSM drivers, no PiFinder clone or venv.
\end{itemize}
\vspace{0.2em}

\begin{frame}[t]{PiFinder LX200}
\begin{columns}[T]
\column{0.34\textwidth}
\begin{itemize}
\item Talks LX200 to PiFinder's \code{pos\_server.py} (TCP 4030).
\item Reports the solved position each poll; a GoTo is only a push-to.
\item No Sync — PiFinder measures the truth every frame.
\end{itemize}
\column{0.64\textwidth}
\hfill\includegraphics[width=\linewidth,height=0.74\textheight,keepaspectratio]{lx200_cp}

\begin{frame}[t]{The Coupling Dial}
\begin{columns}[T]
\column{0.44\textwidth}
\begin{enumerate}
\item \textbf{Off}\\ pure push-to
\item \textbf{Verify / Alert only}\\ warn on drift, hands off
\item \textbf{Auto-correct on drift}\\ Sync past the threshold
\item \textbf{Goto-Forward}\\ real Goto, then settle \& refine

\end{itemize}

\subsection{Upstream: PiFinder}

\begin{frame}[t]{PiFinder — patches that aren't PFSM-specific}
\code{docs/upstream\_patch\_inventory.md}
\begin{itemize}
\item \textbf{Multi-client \code{pos\_server.py}} — a thread per connection. It served one client at a time; SkySafari, Stellarium, KStars and our LX200 driver all want it at once. \code{(docs/concepts/pos\_server\_multi\_client\_architecture.md)}
\item Test Mode toggle + status readback — a reliable API trigger.
\item Camera-process resilience — fall back to the debug camera instead of crashing.
\item \code{all\_ips()} — show every network address, not just the default route.
\item LX200 ``no position yet'' — return \code{None}, not a parseable fake coordinate.
\item Near-pole \code{/api/fake\_solve} RA — filed as \code{brickbots/PiFinder\#645}.
\end{itemize}
Ready-to-file PR text: \code{docs/upstream\_pr\_templates.md}.
\end{frame}

\subsection{App: StellarMate}

\begin{frame}[t]{StellarMate — extending the SMOS app for PFSM}
What a thin integration needs:
\begin{itemize}
\item Start / stop the selected INDI profile.
\item Add / remove just the PiFinder drivers.
\item Pick ``the mount'' from the loaded drivers.
\item Three Coupling presets, in PiFinder's words.
\end{itemize}
The coupling logic is already framework-agnostic Python — a thin adapter ports it.\\
Concept: \code{docs/concepts/mount\_bridge\_web\_integration.md}.
\end{frame}
